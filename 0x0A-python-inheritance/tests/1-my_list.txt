"""Tests for a class that takes a list and sorts it"""

Import class:
	>>> MyList = __import__('1-my_list').MyList

Check for correct instantiation:
	>>> mylist = MyList()
	>>> mylist
	[]

Check for a randomized list:
	>>> l1 = [3, 1, 9, 8, 6]
	>>> sort = MyList(l1)
	>>> sort.print_sorted()
	[1, 3, 6, 8, 9]

Check for empty list:
	>>> l1 = []
	>>> sort = MyList(l1)
	>>> sort.print_sorted()
	[]

Check for negative numbers:
	>>> l1 = [-2, -7, -9, -4, -3, -2]
	>>> sort = MyList(l1)
	>>> sort.print_sorted()
	[-9, -7, -4, -3, -2, -2]

Check for both positive and negative numbers:
	>>> l1 = [4, -2, 7, 9, 5, -5, -2, 10, -34]
	>>> sort = MyList(l1)
	>>> sort.print_sorted()
	[-34, -5, -2, -2, 4, 5, 7, 9, 10]

Check for floats:
	>>> l1 = [3.2, 8,9, 1.2, 0.5, 8.7]
	>>> mylist = MyList(l1)
	>>> mylist.print_sorted()
	[0.5, 1.2, 3.2, 8, 8.7, 9]
